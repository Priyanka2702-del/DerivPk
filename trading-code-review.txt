function money(value: number | string | undefined) {
  const number = Number(value ?? 0);

  return new Intl.NumberFormat("en-US", {
    style: "currency",
    currency: "USD",
    minimumFractionDigits: 2,
    maximumFractionDigits: 2,
  }).format(Number.isFinite(number) ? number : 0);
}

function number(value: number | string | undefined) {
  const n = Number(value ?? 0);

  return Number.isFinite(n) ? n.toFixed(2) : "0.00";
}

      setError("Enter an amount between $0.01 and $1,000.");
      return;
    }

    try {
      setFunding(true);

      const response = await fetch("/api/trading/demo/fund", {
        method: "POST",
        headers: { "Content-Type": "application/json" },
        body: JSON.stringify({ amount }),
      });

      const result = await readResponse(response);

      if (!response.ok || !result?.success) {
        throw new Error(result?.message || "Unable to add demo funds.");
      }

      if (result.fundingComplete) {
        setSuccess(
          `${money(amount)} demo funds were added to your account.`
        );
      } else {
        setSuccess(
          result.message ||
            "Funding submitted. Refresh shortly to check the account balance."
        );
      }

      await loadData();
    } catch (err) {
      console.error("DEMO FUNDING ERROR:", err);

    event.preventDefault();
    setError("");
    setSuccess("");

    if (!data?.account) {
      setError("Demo account information is not available.");
      return;

    const cleanSymbol = symbol.trim().toUpperCase();

    const useAmount = type === "market" && orderMode === "amount";
    const quantity = Number(qty);

function getStatusFromAlpacaError(message: string): number {
  const lower = message.toLowerCase();

  if (message.includes("404")) return 404;
  if (message.includes("409") || lower.includes("conflict")) return 409;
  if (
    message.includes("422") ||
    lower.includes("unprocessable")
  ) return 422;
  if (
    message.includes("403") ||
    lower.includes("forbidden")
  ) return 403;
  if (
    message.includes("401") ||
    lower.includes("unauthorized")
  ) return 401;
  if (
    message.includes("400") ||
    lower.includes("bad request")
  ) return 400;

  return 502;
}

/**
 * POST
 * Supports:
 * - Market orders by dollar amount: { notional: "25.00" }
 * - Orders by share quantity: { qty: "1" }
 * - Limit orders by quantity and limitPrice
 */
export async function POST(request: NextRequest) {
  try {
    const user = await requireAuth();

    if (!user) {
      return NextResponse.json(
        { success: false, message: "Unauthorized" },
        { status: 401 }
      );
    }

    let body: {
      symbol?: string;
      side?: OrderSide;
      type?: OrderType;
      timeInForce?: TimeInForce;
      qty?: number | string;
      notional?: number | string;
      limitPrice?: number | string;
    };

    try {
      body = await request.json();
    } catch {
      return NextResponse.json(
        {
          success: false,
          message: "Invalid JSON request body.",
          code: "INVALID_JSON",
        },
        { status: 400 }
      );
    }

    const symbol = normalizeSymbol(body.symbol);
    const side = body.side;
    const type = body.type;
    const timeInForce = body.timeInForce ?? "day";

    const hasNotional =
      body.notional !== undefined &&
      body.notional !== null &&
      body.notional !== "";

    const amount = hasNotional ? Number(body.notional) : undefined;
    const qty = hasNotional ? undefined : Number(body.qty);

    const limitPrice =
      body.limitPrice === undefined ||
      body.limitPrice === null ||
      body.limitPrice === ""
        ? undefined
        : Number(body.limitPrice);

    // Basic request validation
    if (!/^[A-Z][A-Z0-9.-]{0,19}$/.test(symbol)) {
      return NextResponse.json(
        {
          success: false,
          message: "Enter a valid stock symbol.",
          code: "INVALID_SYMBOL",
        },
        { status: 400 }
      );
    }

    if (side !== "buy" && side !== "sell") {
      return NextResponse.json(
        {
          success: false,
          message: "Side must be either buy or sell.",
          code: "INVALID_SIDE",
        },
        { status: 400 }
      );
    }

    if (type !== "market" && type !== "limit") {
      return NextResponse.json(
        {
          success: false,
          message: "Order type must be market or limit.",
          code: "INVALID_ORDER_TYPE",
        },
        { status: 400 }
      );
    }

    if (timeInForce !== "day" && timeInForce !== "gtc") {
      return NextResponse.json(
        {
          success: false,
          message: "Time in force must be day or gtc.",
          code: "INVALID_TIME_IN_FORCE",
        },
        { status: 400 }
      );
    }

    // Dollar-amount orders are market/day orders only.
    if (hasNotional && (type !== "market" || timeInForce !== "day")) {
      return NextResponse.json(
        {
          success: false,
          message:
            "Dollar-amount orders are supported for market orders with day time-in-force only.",
          code: "NOTIONAL_ORDER_NOT_SUPPORTED",
        },
        { status: 400 }
      );
    }

    if (hasNotional) {
      if (
        !Number.isFinite(amount) ||
        amount! < 1 ||
        Math.abs(amount! * 100 - Math.round(amount! * 100)) > 1e-7
      ) {
        return NextResponse.json(
          {
            success: false,
            message:
              "Order amount must be at least $1.00 and have at most two decimal places.",
            code: "INVALID_NOTIONAL",
          },
          { status: 400 }
        );
      }
    } else if (!Number.isFinite(qty) || qty! <= 0) {
      return NextResponse.json(
        {
          success: false,
          message: "Quantity must be greater than 0.",
          code: "INVALID_QUANTITY",
        },
        { status: 400 }
      );
    }

    if (
      type === "limit" &&
      (!Number.isFinite(limitPrice) || limitPrice! <= 0)
    ) {
      return NextResponse.json(
        {
          success: false,
          message: "A valid limit price is required for a limit order.",
          code: "INVALID_LIMIT_PRICE",
        },
        { status: 400 }
      );
    }

    await connectDB();

    // Always resolve the account from the authenticated user's database record.
    const tradingAccount = await TradingAccount.findOne({
      userId: user.id,
      provider: "alpaca",
      accountType: "paper",
    }).lean();

    if (!tradingAccount) {
      return NextResponse.json(
        {
          success: false,
          message: "Paper trading account not found.",

        lowerMessage.includes("closed")
      ) {
        userMessage = "The market is currently closed for this order.";
        code = "MARKET_CLOSED";

          success: false,
          message: userMessage,
          error: message,
          code,
          symbol,
          side,
          type,
          notional: hasNotional ? amount : undefined,
          qty: hasNotional ? undefined : qty,

/**
 * Convert a MongoDB Date/string into Alpaca's required:
 *
 * YYYY-MM-DD
 *
 * We intentionally use UTC getters so that timezone conversion does not
 * unexpectedly change the user's calendar date.
 */
function formatDateOfBirth(value: Date | string | unknown): string {
  if (!value) {
    throw new Error("Date of birth is required");
  }

  const date = value instanceof Date ? value : new Date(String(value));

  if (Number.isNaN(date.getTime())) {
    throw new Error("Invalid date of birth");
  }

  const year = date.getUTCFullYear();
  const month = String(date.getUTCMonth() + 1).padStart(2, "0");
  const day = String(date.getUTCDate()).padStart(2, "0");

  const formatted = `${year}-${month}-${day}`;

  // Final strict YYYY-MM-DD validation.
  if (!/^\d{4}-\d{2}-\d{2}$/.test(formatted)) {
    throw new Error("Invalid date of birth format");
  }

  return formatted;
}

          success: false,
          message:
            "Trading profile not found. Please complete the account application first.",
        },
        { status: 400 }
      );
    }

    /* ---------------------------------------------------------------------- */
    /* Prevent duplicate account                                              */
    /* ---------------------------------------------------------------------- */

    const existingAccount =
      await TradingAccount.findOne({
        userId: user.id,
        accountType: requestedAccountType,
      }).lean();

    if (existingAccount) {
      return NextResponse.json(
        {
          success: false,
          message: `You already have a ${requestedAccountType} trading account.`,
          account: {
            id: existingAccount.providerAccountId,
            providerAccountId:
              existingAccount.providerAccountId,
            provider: existingAccount.provider,
            accountType:
              existingAccount.accountType,
            status: existingAccount.status,
          },
        },
        { status: 409 }
      );
    }

    /* ---------------------------------------------------------------------- */
    /* Validate profile fields                                                */
    /* ---------------------------------------------------------------------- */

    const givenName = requiredString(
      tradingProfile.givenName,
      "Given name"
    );

    const familyName = requiredString(
      tradingProfile.familyName,
      "Family name"
    );

    const phoneNumber = requiredString(
      tradingProfile.phoneNumber,
      "Phone number"
    );

    const city = requiredString(
      tradingProfile.city,
      "City"
    );

    const state = requiredString(
      tradingProfile.state,
      "State"
    );

    const postalCode = requiredString(
      tradingProfile.postalCode,
      "Postal code"
    );

    const country = requiredString(
      tradingProfile.country,
      "Country"
    ).toUpperCase();

    const countryOfTaxResidence =
      requiredString(
        tradingProfile.countryOfTaxResidence,
        "Country of tax residence"
      ).toUpperCase();

    if (!SUPPORTED_COUNTRIES.has(country)) {
      return NextResponse.json(
        {
          success: false,
          message:
            "Country must be a valid supported ISO alpha-3 country code.",
        },
        { status: 422 }
      );
    }

    if (
      !SUPPORTED_COUNTRIES.has(
        countryOfTaxResidence
      )
    ) {
      return NextResponse.json(
        {
          success: false,
          message:
            "Country of tax residence must be a valid supported ISO alpha-3 country code.",
        },
        { status: 422 }
      );
    }

    /* ---------------------------------------------------------------------- */
    /* Street address                                                         */
    /* ---------------------------------------------------------------------- */

    if (
      !Array.isArray(tradingProfile.streetAddress) ||
      tradingProfile.streetAddress.length === 0 ||
      tradingProfile.streetAddress.some(
        (item) =>
          typeof item !== "string" ||
          !item.trim()
      )
    ) {
      return NextResponse.json(
        {
          success: false,
          message: "Street address is required.",
        },
        { status: 422 }
      );
    }

    const streetAddress =
      tradingProfile.streetAddress.map(
        (item) => item.trim()
      );

    /* ---------------------------------------------------------------------- */
    /* Date of birth                                                          */
    /* ---------------------------------------------------------------------- */

    let dateOfBirth: string;

    try {
      dateOfBirth = formatDateOfBirth(
        tradingProfile.dateOfBirth
      );
    } catch (error) {
      return NextResponse.json(
        {
          success: false,
          message: getErrorMessage(error),
        },
        { status: 422 }
      );
    }

    /*
     * Debug information intentionally does NOT print the user's full
     * date of birth. It only confirms the final format.
     */
    console.log(
      "ALPACA DOB FORMAT:",
      dateOfBirth.replace(/\d/g, "X")
    );

    /* ---------------------------------------------------------------------- */
    /* Tax information                                                        */
    /* ---------------------------------------------------------------------- */

    const taxId =
      typeof tradingProfile.taxId === "string"
        ? tradingProfile.taxId.trim()
        : "";

    const taxIdType =
      typeof tradingProfile.taxIdType === "string"
        ? tradingProfile.taxIdType.trim()
        : "";

    if (!taxId) {
      return NextResponse.json(
        {
          success: false,
          message:
            "Tax ID is required to create a trading account.",
        },
        { status: 422 }
      );
    }

    if (!taxIdType) {
      return NextResponse.json(
        {
          success: false,
          message:
            "Tax ID type is required to create a trading account.",
        },
        { status: 422 }
      );
    }

    if (!/[\d]/.test(taxId)) {
      return NextResponse.json(
        {
          success: false,
          message:
            "Tax ID must contain at least one digit.",
        },
        { status: 422 }
      );
    }

    const normalizedTaxResidence =
      countryOfTaxResidence;

    /* ---------------------------------------------------------------------- */
    /* Employment                                                             */
    /* ---------------------------------------------------------------------- */

    const allowedEmploymentStatuses = [
      "EMPLOYED",
      "UNEMPLOYED",
      "RETIRED",
      "STUDENT",
    ];

    const employmentStatus =
      typeof tradingProfile.employmentStatus ===
      "string"
        ? tradingProfile.employmentStatus
            .trim()
            .toUpperCase()
        : "";

    if (
      !allowedEmploymentStatuses.includes(
        employmentStatus
      )
    ) {
      return NextResponse.json(
        {
          success: false,
          message:
            "Employment status must be one of: EMPLOYED, UNEMPLOYED, RETIRED, STUDENT.",
        },
        { status: 422 }
      );
    }

    /* ---------------------------------------------------------------------- */
    /* Funding source                                                         */
    /* ---------------------------------------------------------------------- */

    if (
      !Array.isArray(tradingProfile.fundingSource) ||
      tradingProfile.fundingSource.length === 0
    ) {
      return NextResponse.json(
        {
          success: false,
          message:
            "At least one funding source is required.",
        },
        { status: 422 }
      );
    }

    const fundingSource =
      tradingProfile.fundingSource
        .filter(
          (item) =>
            typeof item === "string" &&
            item.trim()
        )
        .map((item) => item.trim());

    if (fundingSource.length === 0) {
      return NextResponse.json(
        {
          success: false,
          message:
            "At least one valid funding source is required.",
        },
        { status: 422 }
      );
    }

    /* ---------------------------------------------------------------------- */
    /* Risk tolerance                                                         */
    /* ---------------------------------------------------------------------- */

    const allowedRiskTolerance = [
      "conservative",
      "moderate",
      "significant_risk",
    ];

    if (
      !allowedRiskTolerance.includes(
        tradingProfile.riskTolerance
      )
    ) {
      return NextResponse.json(
        {
          success: false,
          message:
            "Invalid risk tolerance.",
        },
        { status: 422 }
      );
    }

    /* ---------------------------------------------------------------------- */
    /* Disclosures                                                            */
    /* ---------------------------------------------------------------------- */

    if (!tradingProfile.disclosures) {
      return NextResponse.json(
        {
          success: false,
          message:
            "Required disclosures are missing.",
        },
        { status: 422 }
      );
    }

    const disclosures = {
      is_control_person:
        Boolean(
          tradingProfile.disclosures
            .isControlPerson
        ),

      is_affiliated_exchange_or_finra:
        Boolean(
          tradingProfile.disclosures
            .isAffiliatedExchangeOrFinra
        ),

      is_politically_exposed:
        Boolean(
          tradingProfile.disclosures
            .isPoliticallyExposed
        ),

      immediate_family_exposed:
        Boolean(
          tradingProfile.disclosures
            .immediateFamilyExposed
        ),

      employment_status:
        employmentStatus,
    };

    /* ---------------------------------------------------------------------- */
    /* Agreements                                                             */
    /* ---------------------------------------------------------------------- */

    if (!tradingProfile.agreementsAccepted) {
      return NextResponse.json(
        {
          success: false,
          message:
            "You must accept the required agreements before creating a trading account.",
        },
        { status: 422 }
      );
    }

    const signedAt =
      new Date().toISOString();

    const ipAddress =
      getRequestIp(request);

    const agreements = [
      {
        agreement: "account_agreement",
        signed_at: signedAt,
        ip_address: ipAddress,
      },
      {
        agreement: "customer_agreement",
        signed_at: signedAt,
        ip_address: ipAddress,
      },
      {
        agreement: "margin_agreement",
        signed_at: signedAt,
        ip_address: ipAddress,
      },
    ];

    /* ---------------------------------------------------------------------- */
    /* Logging                                                                */
    /* ---------------------------------------------------------------------- */

    console.log(
      "CREATING ALPACA ACCOUNT FOR USER:",
      {
        userId: user.id,
        email: dbUser.email,
        applicationAccountType:
          requestedAccountType,
        country,
        countryOfTaxResidence:
          normalizedTaxResidence,
        agreementTypes:
          agreements.map(
            (agreement) =>
              agreement.agreement
          ),
      }
    );

    /* ---------------------------------------------------------------------- */
    /* Alpaca account payload                                                 */
    /* ---------------------------------------------------------------------- */

    const alpacaPayload = {
      account_type: "trading",

      contact: {
        email_address: dbUser.email,

        phone_number: phoneNumber,

        street_address:
          streetAddress,

        city,

        state,

        postal_code:
          postalCode,

        country,
      },

      identity: {
        given_name: givenName,

        family_name: familyName,

        /*
         * IMPORTANT:
         * Alpaca expects YYYY-MM-DD.
         *
         * Example:
         * 2003-11-03
         */
        date_of_birth:
          dateOfBirth,

        tax_id: taxId,

        tax_id_type:
          taxIdType,

        country_of_tax_residence:
          normalizedTaxResidence,

        funding_source:
          fundingSource,
      },

      disclosures,

      risk_tolerance:
        tradingProfile.riskTolerance,

      agreements,
    };

    /* ---------------------------------------------------------------------- */
    /* Create account at Alpaca                                               */
    /* ---------------------------------------------------------------------- */

    const alpacaAccount =
      await alpacaRequest<CreateAccountResponse>(
        "/v1/accounts",
        {
          method: "POST",

          body: JSON.stringify(
            alpacaPayload
          ),
        }
      );

    if (!alpacaAccount?.id) {
      throw new Error(
        "Alpaca did not return a valid account ID."
      );
    }

    /* ---------------------------------------------------------------------- */
    /* Save account in MongoDB                                                */
    /* ---------------------------------------------------------------------- */

    const savedAccount =
      await TradingAccount.create({
        userId: user.id,

        provider: "alpaca",

        providerAccountId:
          alpacaAccount.id,

        accountType:
          requestedAccountType,

        status:
          alpacaAccount.status ||
          "SUBMITTED",
      });

    /* ---------------------------------------------------------------------- */
    /* Response                                                               */
    /* ---------------------------------------------------------------------- */

    return NextResponse.json(
      {
        success: true,

        message:
          "Trading account created successfully.",

        account: {
          id: alpacaAccount.id,

          providerAccountId:
            alpacaAccount.id,

          provider: "alpaca",

          accountType:
            savedAccount.accountType,

          status:
            alpacaAccount.status ||
            savedAccount.status ||
            "SUBMITTED",

          accountNumber:
            alpacaAccount.account_number ||
            null,

          currency:
            alpacaAccount.currency ||
            "USD",

          cash:
            alpacaAccount.cash ||
            "0",

          buyingPower:
            alpacaAccount.buying_power ||
            "0",

          portfolioValue:
            alpacaAccount.portfolio_value ||
            "0",

          equity:
            alpacaAccount.equity ||
            "0",
        },
      },
      { status: 201 }
    );
  } catch (error) {
    const message =
      getErrorMessage(error);

    console.error(
      "CREATE TRADING ACCOUNT ERROR:",
      error
    );

    /*
     * Alpaca validation errors are generally 422.
     * Returning 422 to the frontend is more accurate than
     * converting them into a generic 500 error.
     */
    if (
      message.startsWith(
        "422 Unprocessable Entity:"
      )
    ) {
      return NextResponse.json(
        {
          success: false,
          message,
        },
        { status: 422 }
      );
    }

    return NextResponse.json(
      {
        success: false,
        message,
      },
      { status: 500 }
    );
  }
}
