  return NextResponse.json(

  );

export async function PATCH(
  request: Request,
  { params }: { params: Promise<{ requestId: string }> }
) {
  try {
    const admin = await requireAdmin();

    if (!admin) {
      return errorResponse("Admin access required.", 403);
    }

    const { requestId } = await params;

    if (!mongoose.isValidObjectId(requestId)) {
      return errorResponse("Invalid request ID.", 400);
    }

    let body: unknown;

    try {
      body = await request.json();
    } catch {
      return errorResponse("Request body must be valid JSON.");
    }

    if (
      typeof body !== "object" ||
      body === null ||
      Array.isArray(body)
    ) {
      return errorResponse("Invalid request body.");
    }

    const data = body as Record<string, unknown>;
    const action =
      typeof data.action === "string"
        ? data.action.trim().toUpperCase()
        : "";

    const adminNote =
      typeof data.adminNote === "string"
        ? data.adminNote.trim()
        : "";

    const rejectionReason =
      typeof data.rejectionReason === "string"
        ? data.rejectionReason.trim()
        : "";

    if (action !== "APPROVE" && action !== "REJECT") {
      return errorResponse("Action must be APPROVE or REJECT.");
    }

    if (adminNote.length > 1000) {
      return errorResponse("Admin note cannot exceed 1000 characters.");
    }

    if (rejectionReason.length > 1000) {
      return errorResponse("Rejection reason cannot exceed 1000 characters.");
    }

    if (action === "REJECT" && !rejectionReason) {
      return errorResponse("A rejection reason is required.");
    }

    await connectDB();

    const update: Record<string, unknown> = {
      status: action === "APPROVE" ? "APPROVED" : "REJECTED",
      adminDecisionBy: admin.id,
      adminDecisionAt: new Date(),
      adminNote,
    };

    if (action === "REJECT") {
      update.rejectionReason = rejectionReason;
    } else {
      update.rejectionReason = undefined;
    }

    const tradeRequest = await RealTradeRequest.findOneAndUpdate(
      {
        _id: requestId,
        status: "PENDING_REVIEW",
      },
      { $set: update },
      { new: true, runValidators: true }
    ).lean();

    if (!tradeRequest) {
      const existing = await RealTradeRequest.findById(requestId)
        .select("_id status")
        .lean();

      if (!existing) {
        return errorResponse("Trade request not found.", 404);
      }

      return errorResponse(
        "This request has already been reviewed.",
        409
      );
    }

    return NextResponse.json({
      success: true,
      message:
        action === "APPROVE"
          ? "Trade request approved for internal workflow."
          : "Trade request rejected.",
      request: tradeRequest,
      notice:
        "This decision does not execute a market order or process a payment.",
    });
  } catch (error) {
    console.error("ADMIN REAL TRADE DECISION ERROR:", error);
    return errorResponse("Unable to update trade request.", 500);
  }
}

import { NextResponse } from "next/server";

import RealTradeRequest, { type RealTradeStatus } from "@/models/RealTradeRequest";

export const runtime = "nodejs";

function errorResponse(message: string, status = 400) {
  return NextResponse.json(
    { success: false, message },
    { status }
  );
}

export async function GET(request: Request) {
  try {
    const admin = await requireAdmin();

    if (!admin) {
      return errorResponse("Admin access required.", 403);
    }

    const url = new URL(request.url);
    const requestedStatus =
      url.searchParams.get("status")?.toUpperCase() ||
      "PENDING_REVIEW";

    const allowedStatuses = [
      "PENDING_REVIEW",
      "APPROVED",
      "REJECTED",
      "ALL",
    ];

    if (!allowedStatuses.includes(requestedStatus)) {
      return errorResponse(
        "Invalid status filter.",
        400
      );
    }

    const rawLimit = Number(
      url.searchParams.get("limit") || "25"
    );
    const rawPage = Number(
      url.searchParams.get("page") || "1"
    );

    if (
      !Number.isInteger(rawLimit) ||
      rawLimit < 1 ||
      !Number.isInteger(rawPage) ||
      rawPage < 1
    ) {
      return errorResponse(
        "Page and limit must be positive integers."
      );
    }

    const limit = Math.min(rawLimit, 100);
    const page = rawPage;

    await connectDB();

    const filter: { status?: RealTradeStatus } =
      requestedStatus === "ALL"
        ? {}
        : { status: requestedStatus as RealTradeStatus };

    const [requests, total] = await Promise.all([
      RealTradeRequest.find(filter)
        .populate("userId", "name email")
        .populate("accountId", "accountLabel status")
        .sort({ createdAt: -1 })
        .skip((page - 1) * limit)
        .limit(limit)
        .lean(),
      RealTradeRequest.countDocuments(filter),
    ]);

    return NextResponse.json({
      success: true,
      requests,
      pagination: {
        page,
        limit,
        total,
        totalPages: Math.ceil(total / limit),
      },
      notice:
        "Admin review only. Approval does not execute a market order or process payment.",
    });
  } catch (error) {
    console.error("ADMIN REAL TRADE REQUESTS ERROR:", error);

    return errorResponse(
      "Unable to load trade requests.",
      500
    );
  }
}

function isObject(
  value: unknown
): value is Record<string, unknown> {
  return (

      return jsonError(
        "Payment mode is not configured. Set OPEXGT_MODE to mock or live.",
        503
      );
    }

    if (!user) {
      return jsonError("Unauthorized.", 401);
    }

      return jsonError("Invalid user session.", 401);

      body = await request.json();

      return jsonError("Invalid JSON body.", 400);

        400
      );

      !ALLOWED_CURRENCIES.includes(
        requestedCurrency as CryptoCurrency
      )
    ) {
      return jsonError(
        "Currency must be USDT, BTC, or ETH.",
        400
      );

      requestedCurrency as CryptoCurrency;

        currency: "USD",

            status: payment.status,
            provider: payment.provider,
            createdAt: payment.createdAt,

    const providerPaymentId =

        const parsed = new URL(checkoutUrl);

        validCheckoutUrl =

          parsed.hostname === "opexgt.com" &&

      // Reconcile this payment using the provider dashboard.
      return NextResponse.json(
        {
          success: false,
          mode: "live",
          message:
            "OpexGT returned an unexpected response. The payment remains pending and needs reconciliation.",
          paymentId: payment._id.toString(),
          orderId,
          status: "PENDING",
        },
        { status: 502 }
      );
    }

    payment.providerPaymentId = providerPaymentId;
    payment.checkoutUrl = checkoutUrl;
    payment.depositAddress = depositAddress;
    payment.cryptoCurrency = cryptoCurrency;
    payment.cryptoAmount = cryptoAmount;
    payment.expiresAt = expiresAt;

    await payment.save();

    return NextResponse.json(
      {
        success: true,
        mode: "live",
        message:
          "Payment session created. Complete checkout to pay.",
        payment: {
          id: payment._id.toString(),
          orderId,

          purpose: payment.purpose,

          createdAt: payment.createdAt,

    );
  } catch (error) {
    console.error("PAYMENT_CREATE_FAILED", {
      errorName:
        error instanceof Error
          ? error.name
          : "UnknownError",
      message:
        error instanceof Error
          ? error.message
          : "Unknown error",

      "Unable to create payment.",
      500
    );
  }
}

import { NextResponse } from "next/server";
import mongoose from "mongoose";
import { connectDB } from "@/lib/db";
import { requireAuth } from "@/lib/session";
import Payment from "@/models/Payment";
import WalletLedger from "@/models/WalletLedger";

export const runtime = "nodejs";

export async function GET() {
  try {
    // 1. Authenticate the user
    const user = await requireAuth();

    if (!user) {
      return NextResponse.json(
        { error: "Unauthorized" },
        { status: 401 }
      );
    }

    // 2. Validate user ID
    if (!mongoose.isValidObjectId(user.id)) {
      return NextResponse.json(
        { error: "Invalid user session." },
        { status: 401 }
      );
    }

    // 3. Connect to MongoDB
    await connectDB();

    const userId = new mongoose.Types.ObjectId(user.id);

    // 4. Fetch payments, wallet balance and history
    const [payments, ledger, depositTotals, ledgerEntries] =
      await Promise.all([
        Payment.find({ userId })
          .select(
            "purpose amountCents currency status provider providerPaymentId providerOrderId checkoutUrl depositAddress cryptoCurrency cryptoAmount txHash expiresAt tradeRequestId createdAt updatedAt"
          )
          .sort({ createdAt: -1 })
          .limit(50)
          .lean(),

        // Wallet balance comes from ledger entries.
        // Only CREDIT entries increase the displayed balance.
        WalletLedger.aggregate([
          {
            $match: {
              userId,
              currency: "USD",
            },
          },
          {
            $group: {
              _id: null,
              balanceCents: {
                $sum: {
                  $cond: [
                    { $eq: ["$type", "CREDIT"] },
                    "$amountCents",
                    0,
                  ],
                },
              },
            },
          },
        ]),

        // Count only completed USD deposits.
        Payment.aggregate([
          {
            $match: {
              userId,
              purpose: "DEPOSIT",
              status: "COMPLETED",
              currency: "USD",
            },
          },
          {
            $group: {
              _id: null,
              totalDepositedCents: {
                $sum: "$amountCents",
              },
              completedDepositCount: {
                $sum: 1,
              },
            },
          },
        ]),

        // Latest wallet ledger entries
        WalletLedger.find({ userId })
          .select(
            "paymentId type amountCents currency description createdAt"
          )
          .sort({ createdAt: -1 })
          .limit(50)
          .lean(),
      ]);

    // 5. Prepare wallet totals
    const balanceCents = ledger[0]?.balanceCents ?? 0;
    const totalDepositedCents =
      depositTotals[0]?.totalDepositedCents ?? 0;
    const completedDepositCount =
      depositTotals[0]?.completedDepositCount ?? 0;

    // 6. Return payment history and wallet summary
    return NextResponse.json({
      success: true,
      wallet: {
        currency: "USD",
        balanceCents,
        balanceUsd: balanceCents / 100,
        totalDepositedCents,
        totalDepositedUsd: totalDepositedCents / 100,
        completedDepositCount,
      },
      payments,
      ledger: ledgerEntries,
    });
  } catch (error) {
    // Log the real error in the server terminal for debugging.
    console.error("PAYMENT_HISTORY_FETCH_FAILED:", {
      name:
        error instanceof Error ? error.name : "UnknownError",
      message:
        error instanceof Error ? error.message : String(error),
      stack:
        error instanceof Error ? error.stack : undefined,
    });

    return NextResponse.json(
      { error: "Unable to fetch wallet and payment history." },
      { status: 500 }
    );
  }
}

import { createHmac, timingSafeEqual } from "node:crypto";
import { NextResponse } from "next/server";

import { connectDB } from "@/lib/db";
import Payment, { type PaymentStatus } from "@/models/Payment";

export const runtime = "nodejs";

const OPEXGT_STATUS_URL = "https://opexgt.com/api/v1/payment";

function reply(message: string, status: number) {
  return NextResponse.json(
    { success: status >= 200 && status < 300, message },
    { status }
  );
}

function isObject(value: unknown): value is Record<string, unknown> {
  return typeof value === "object" && value !== null && !Array.isArray(value);
}

function verifySignature(
  rawBody: Buffer,
  signature: string | null,
  secret: string
): boolean {
  if (!signature || !signature.startsWith("sha512=")) return false;

  const expected = `sha512=${createHmac("sha512", secret)
    .update(rawBody)
    .digest("hex")}`;

  const expectedBuffer = Buffer.from(expected, "utf8");
  const receivedBuffer = Buffer.from(signature, "utf8");

  return (
    expectedBuffer.length === receivedBuffer.length &&
    timingSafeEqual(expectedBuffer, receivedBuffer)
  );
}

function amountMatches(providerAmount: unknown, amountCents: number) {
  if (
    typeof providerAmount !== "number" ||
    !Number.isFinite(providerAmount)
  ) {
    return false;
  }

  return Math.round(providerAmount * 100) === amountCents;
}

export async function POST(request: Request) {
  try {
    const secret =
      process.env.OPEXGT_SIGNING_SECRET ??
      process.env.OpexGT_Signing_secret;

    if (!secret) {
      console.error("OPEXGT_WEBHOOK_SECRET_MISSING");
      return reply("Webhook is not configured.", 503);
    }

    // Signature must be verified against the original, unmodified body.
    const rawBody = Buffer.from(await request.arrayBuffer());
    const signature = request.headers.get("x-opex-signature");

    if (!verifySignature(rawBody, signature, secret)) {
      return reply("Invalid webhook signature.", 401);
    }

    let payload: unknown;

    try {
      payload = JSON.parse(rawBody.toString("utf8"));
    } catch {
      return reply("Invalid JSON payload.", 400);
    }

    if (!isObject(payload) || typeof payload.event !== "string") {
      return reply("Invalid webhook payload.", 400);
    }

    if (payload.event === "webhook.test") {
      return reply("Webhook signature verified.", 200);
    }

    const allowedEvents = new Set([
      "payment.completed",
      "payment.expired",
      "payment.amount_mismatch",
    ]);

    if (!allowedEvents.has(payload.event)) {
      // Ignore unknown but correctly signed event types.
      return reply("Event ignored.", 200);
    }

    const providerPaymentId = payload.payment_id;
    const orderId = payload.order_id;

    if (
      typeof providerPaymentId !== "string" ||
      !providerPaymentId ||
      typeof orderId !== "string" ||
      !orderId
    ) {
      return reply("Missing payment or order identifier.", 400);
    }

    await connectDB();

    const payment = await Payment.findOne({
      provider: "OPEXGT",
      providerPaymentId,
      providerOrderId: orderId,
    });

    if (!payment) {
      // A provider webhook may arrive before the create route saves its IDs.
      // Return a retryable response instead of silently accepting the event.
      console.error("OPEXGT_WEBHOOK_PAYMENT_NOT_FOUND", {
        providerPaymentId,
        orderId,
      });
      return reply("Payment record not found; retry later.", 503);
    }

    const finalStatuses: PaymentStatus[] = [
      "COMPLETED",
      "EXPIRED",
      "FAILED",
      "CANCELLED",
      "AMOUNT_MISMATCH",
    ];

    if (finalStatuses.includes(payment.status)) {
      return reply("Payment already has a final status.", 200);
    }

    const apiKey =
      process.env.OPEXGT_API_KEY ??
      process.env.OpexGT_Lve_key;

    if (!apiKey) {
      console.error("OPEXGT_STATUS_API_KEY_MISSING");
      return reply("Payment status verification is not configured.", 503);
    }

    let statusResponse: Response;

    try {
      statusResponse = await fetch(
        `${OPEXGT_STATUS_URL}/${encodeURIComponent(providerPaymentId)}`,
        {
          method: "GET",
          headers: {
            Authorization: `Bearer ${apiKey}`,
            Accept: "application/json",
          },
          cache: "no-store",
          signal: AbortSignal.timeout(10_000),
        }
      );
    } catch (error) {
      console.error("OPEXGT_STATUS_LOOKUP_FAILED", {
        paymentId: payment._id.toString(),
        errorName: error instanceof Error ? error.name : "UnknownError",
      });

      return reply("Could not verify provider status; retry later.", 503);
    }

    if (!statusResponse.ok) {
      console.error("OPEXGT_STATUS_LOOKUP_REJECTED", {
        paymentId: payment._id.toString(),
        httpStatus: statusResponse.status,
      });

      return reply("Provider status verification failed; retry later.", 503);
    }

    let statusPayload: unknown;

    try {
      statusPayload = await statusResponse.json();
    } catch {
      return reply("Provider returned invalid status data.", 503);
    }

    if (!isObject(statusPayload)) {
      return reply("Provider returned invalid status data.", 503);
    }

    const providerData = isObject(statusPayload.data)
      ? statusPayload.data
      : statusPayload;

    const confirmedPaymentId =
      typeof providerData.id === "string"
        ? providerData.id
        : typeof providerData.payment_id === "string"
          ? providerData.payment_id
          : null;

    const confirmedOrderId =
      typeof providerData.order_id === "string"
        ? providerData.order_id
        : null;

    const confirmedStatus =
      typeof providerData.status === "string"
        ? providerData.status.toLowerCase()
        : "";

    const confirmedCurrency =
      typeof providerData.currency === "string"
        ? providerData.currency.toUpperCase()
        : "";

    // Per the supplied API example, "amount" is assumed to be the USD order
    // amount, while "currency" is assumed to be the selected crypto currency.
    // Confirm these field meanings with OpexGT before enabling live payments.
    const confirmedAmount = providerData.amount;

    const identifiersMatch =
      confirmedPaymentId === providerPaymentId &&
      confirmedOrderId === orderId;

    if (!identifiersMatch) {
      console.error("OPEXGT_STATUS_IDENTIFIER_MISMATCH", {
        paymentId: payment._id.toString(),
        paymentIdMatched: confirmedPaymentId === providerPaymentId,
        orderIdMatched: confirmedOrderId === orderId,
      });

      // Unexpected data is not automatically treated as an actual
      // amount mismatch. Leave the payment pending for investigation.
      return reply("Provider identifiers did not match; review required.", 503);
    }

    if (
      confirmedCurrency !== payment.cryptoCurrency ||
      !amountMatches(confirmedAmount, payment.amountCents)
    ) {
      console.error("OPEXGT_STATUS_AMOUNT_OR_CURRENCY_MISMATCH", {
        paymentId: payment._id.toString(),
        currencyMatched: confirmedCurrency === payment.cryptoCurrency,
        amountMatched: amountMatches(confirmedAmount, payment.amountCents),
      });

      // Do not mark as completed or guess what happened. Keep it pending
      // until the provider's amount/currency fields are confirmed.
      return reply("Provider amount or currency needs reconciliation.", 503);
    }

    let nextStatus: PaymentStatus;

    if (payload.event === "payment.amount_mismatch") {
      if (
        confirmedStatus !== "pending" &&
        confirmedStatus !== "completed"
      ) {
        return reply("Provider status requires manual reconciliation.", 503);
      }

      // Per the supplied docs, amount mismatch needs manual handling
      // in OpexGT. Never auto-complete this event.
      nextStatus = "AMOUNT_MISMATCH";
    } else if (payload.event === "payment.completed") {
      if (confirmedStatus !== "completed") {
        return reply(
          "Provider has not confirmed payment completion; retry later.",
          503
        );
      }

      nextStatus = "COMPLETED";
    } else {
      if (confirmedStatus !== "expired") {
        return reply(
          "Provider has not confirmed payment expiry; retry later.",
          503
        );
      }

      nextStatus = "EXPIRED";
    }

    const txHash =
      typeof providerData.tx_hash === "string"
        ? providerData.tx_hash
        : typeof payload.tx_hash === "string"
          ? payload.tx_hash
          : undefined;

    const update: { status: PaymentStatus; txHash?: string } = {
      status: nextStatus,
    };

    if (txHash) {
      update.txHash = txHash;
    }

    const result = await Payment.updateOne(
      { _id: payment._id, status: "PENDING" },
      { $set: update }
    );

    if (result.modifiedCount === 0) {
      const current = await Payment.findById(payment._id)
        .select("status")
        .lean();

      if (
        current &&
        finalStatuses.includes(current.status as PaymentStatus)
      ) {
        return reply("Payment was already processed.", 200);
      }

      return reply("Payment status changed concurrently; retry later.", 503);
    }

    // This updates the payment record only. It does not credit a wallet,
    // approve a trade request, or execute a trade.
    return reply("Webhook processed.", 200);
  } catch (error) {
    console.error("OPEXGT_WEBHOOK_PROCESSING_FAILED", {
      errorName: error instanceof Error ? error.name : "UnknownError",
      message: error instanceof Error ? error.message : "Unknown error",
    });

    return reply("Unable to process webhook.", 500);
  }
}

import mongoose from "mongoose";

type MongooseCache = {
  conn: typeof mongoose | null;
  promise: Promise<typeof mongoose> | null;
};

declare global {
  var mongooseCache:
    | MongooseCache
    | undefined;
}

const cached: MongooseCache =
  global.mongooseCache ?? {
    conn: null,
    promise: null,
  };

global.mongooseCache = cached;

export async function connectDB() {
  const MONGODB_URI =
    process.env.MONGODB_URI;

  if (!MONGODB_URI) {
    throw new Error(
      "Please define MONGODB_URI in environment variables."
    );
  }

  if (cached.conn) {
    return cached.conn;
  }

  if (!cached.promise) {
    cached.promise = mongoose.connect(
      MONGODB_URI,
      {
        maxPoolSize: 10,
        minPoolSize: 2,
        serverSelectionTimeoutMS: 10000,
        socketTimeoutMS: 45000,
        family: 4,
      }
    );
  }

  try {
    cached.conn =
      await cached.promise;
  } catch (error) {
    cached.promise = null;
    throw error;
  }

  return cached.conn;
}

import crypto from "crypto";

const DERIV_AUTH_URL = "https://auth.deriv.com/oauth2/auth";
const DERIV_TOKEN_URL = "https://auth.deriv.com/oauth2/token";

const CLIENT_ID = process.env.DERIV_CLIENT_ID;
const REDIRECT_URI = process.env.DERIV_REDIRECT_URI;

if (!CLIENT_ID) {
  throw new Error("DERIV_CLIENT_ID is not defined");
}

if (!REDIRECT_URI) {
  throw new Error("DERIV_REDIRECT_URI is not defined");
}

/**
 * Generate a cryptographically secure random string.
 */
export function generateCodeVerifier(): string {
  return crypto.randomBytes(64).toString("base64url");
}

/**
 * Generate PKCE code_challenge from code_verifier.
 */
export function generateCodeChallenge(
  codeVerifier: string
): string {
  return crypto
    .createHash("sha256")
    .update(codeVerifier)
    .digest("base64url");
}

/**
 * Generate OAuth state for CSRF protection.
 */
export function generateOAuthState(): string {
  return crypto.randomBytes(32).toString("hex");
}

/**
 * Build Deriv OAuth authorization URL.
 */
export function buildDerivAuthorizationUrl(
  state: string,
  codeChallenge: string
): string {
  const params = new URLSearchParams({
    response_type: "code",
    client_id: CLIENT_ID!,
    redirect_uri: REDIRECT_URI!,
    scope: "trade",
    state,
    code_challenge: codeChallenge,
    code_challenge_method: "S256",
  });

  return `${DERIV_AUTH_URL}?${params.toString()}`;
}

/**
 * Exchange authorization code for Deriv access token.
 */
export async function exchangeCodeForToken(
  code: string,
  codeVerifier: string
) {
  const body = new URLSearchParams({
    grant_type: "authorization_code",
    client_id: CLIENT_ID!,
    code,
    code_verifier: codeVerifier,
    redirect_uri: REDIRECT_URI!,
  });

  const response = await fetch(DERIV_TOKEN_URL, {
    method: "POST",
    headers: {
      "Content-Type":
        "application/x-www-form-urlencoded",
    },
    body: body.toString(),
    cache: "no-store",
  });

  const data = await response.json();

  if (!response.ok) {
    console.error(
      "Deriv token exchange failed:",
      response.status,
      data
    );

    throw new Error(
      data?.error_description ||
        data?.error ||
        "Failed to exchange authorization code"
    );
  }

  return data;
}
